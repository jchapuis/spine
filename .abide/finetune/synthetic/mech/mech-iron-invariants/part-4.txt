import io.github.iltotore.iron.*
import io.github.iltotore.iron.constraint.numeric.Interval
import org.http4s.Headers

object RetryPolicy:
  val defaultAttempts: Int :| Interval.Closed[1, 10] = 3

  def fromHeaders(headers: Headers): Int :| Interval.Closed[1, 10] =
    headers.get(org.typelevel.ci.CIString("X-Retry")).map(_.head.value.toInt).fold(defaultAttempts)(
      _.refineUnsafe[Interval.Closed[1, 10]]
    )

object Versions:
  def parse(raw: String): Option[Version] =
    raw.split('.').toList.traverse(_.toIntOption).collect:
      case major :: minor :: patch :: Nil => Version(major, minor, patch)

  def bump(current: Version, next: Version): Version =
    require(next > current, s"$next is not newer than $current")
    next

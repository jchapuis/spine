package com.gatekeep.auth

import cats.data.{Kleisli, OptionT}
import cats.effect.IO
import cats.effect.std.UUIDGen
import org.http4s.{HttpRoutes, Request}
import org.typelevel.ci.*

object RequestIdMiddleware:

  def apply(routes: HttpRoutes[IO]): HttpRoutes[IO] =
    Kleisli: (request: Request[IO]) =>
      val existing = request.headers.get(ci"X-Request-Id").map(_.head.value)
      OptionT.liftF(existing.fold(UUIDGen[IO].randomUUID.map(_.toString))(IO.pure)).flatMap: id =>
        routes(request).map(_.putHeaders("X-Request-Id" -> id))

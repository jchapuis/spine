object SlotSearch:
  def contains(slots: Vector[LocalTime], time: LocalTime): Boolean =
    slots.contains(time)

  def firstAtOrAfter(slots: Vector[LocalTime], time: LocalTime): Option[LocalTime] =
    var low = 0
    var high = slots.length

    slots.lift(low)

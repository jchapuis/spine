object DistanceMatrix:
  def size(stops: Vector[String]): Int = stops.size * stops.size

  def build(stops: Vector[String], distance: (String, String) => Double): Array[Array[Double]] =
    val matrix = Array.ofDim[Double](stops.size, stops.size)
    for row <- stops.indices do
      var column = 0
      while column < stops.size do
        matrix(row)(column) = distance(stops(row), stops(column))
        column += 1
    matrix

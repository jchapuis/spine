package com.kestrel.auth

import java.time.Duration
import java.time.Instant
import munit.FunSuite

class SessionExpirySuite extends FunSuite:
  test("session older than its ttl is expired"):
    val openedAt = Instant.parse("2026-01-01T00:00:00Z")
    val session = Session(SessionId("s-1"), UserId("u-1"), openedAt)
    assert(SessionExpiry.isExpired(session, openedAt.plus(Duration.ofHours(2)), Duration.ofHours(1)))

  test("fresh session is not expired"):
    val openedAt = Instant.parse("2026-01-01T00:00:00Z")
    val session = Session(SessionId("s-2"), UserId("u-2"), openedAt)
    assert(!SessionExpiry.isExpired(session, openedAt.plusSeconds(30), Duration.ofHours(1)))

import cats.effect.IO
import cats.effect.Ref

class JobRunner(states: Ref[IO, Map[JobId, JobState]]):
  def start(id: JobId): IO[Unit] =
    states.update(_.updated(id, JobState.Running))

  def finish(id: JobId, ok: Boolean): IO[Unit] =
    states.update(_.updated(id, if ok then JobState.Succeeded else JobState.Failed))

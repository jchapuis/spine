package com.acme.notifications

import cats.effect.IO
import scala.concurrent.duration.*

object Retry:
  def once[A](action: IO[A]): IO[A] =
    action.handleErrorWith(_ => IO.sleep(500.millis) *> action)

  def times[A](action: IO[A], attempts: Int, delay: FiniteDuration): IO[A] =
    if attempts <= 1 then action
    else action.handleErrorWith(_ => IO.sleep(delay) *> times(action, attempts - 1, delay * 2))

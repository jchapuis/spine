import scala.util.matching.Regex

object QueryParser:
  private val quoted: Regex = "\"([^\"]+)\"".r

  def terms(query: String): List[String] =
    val phrases = quoted.findAllMatchIn(query).map(_.group(1)).toList

    phrases ++ rest.split("\\s+").toList.filter(_.nonEmpty)

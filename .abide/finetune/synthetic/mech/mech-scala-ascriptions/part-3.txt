class PriceBook(prices: List[Price]):
  def forRegion(region: String): List[Price] =
    prices.filter(_.region == region)

  def cheapest(region: String): Map[String, Long] =
    lowestBySku(forRegion(region), discountPercent = 0)

  private def lowestBySku(
      regional: List[Price],

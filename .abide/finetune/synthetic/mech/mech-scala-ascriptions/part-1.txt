import cats.effect.Clock
import cats.effect.IO

object CardValidator:
  def validate(number: String, expiryYear: Int)(using Clock[IO]): IO[Boolean] =
    Clock[IO].realTimeInstant.map(_ => luhn(number).isRight && expiryYear > 2000)

  private def luhn(number: String): Either[String, String] =
    val digits = number.filter(_.isDigit).map(_.asDigit).reverse
    val sum = digits.zipWithIndex.map((d, i) => if i % 2 == 1 then (d * 2) % 9 else d).sum
    Either.cond(sum % 10 == 0, number, s"checksum failed for $number")
